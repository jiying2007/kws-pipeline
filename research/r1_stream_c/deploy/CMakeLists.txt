cmake_minimum_required(VERSION 3.16)
project(kws_r1_reference64 VERSION 0.1.0 LANGUAGES C)
if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "This research build is verified only with GCC/Clang")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(GNUInstallDirs)
set(CORE "${CMAKE_CURRENT_SOURCE_DIR}/reference64/source")
set(ADAPTER "${CMAKE_CURRENT_SOURCE_DIR}/adapter")
set(SOURCES "${ADAPTER}/r1_sdk.c" "${CORE}/r1d_stream.c" "${CORE}/r1d_pcm.c"
    "${CORE}/reference64_fbank.c" "${CORE}/r1dfsmn.c" "${CORE}/sha256.c")
add_library(r1_objects OBJECT ${SOURCES})
set_target_properties(r1_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(r1_objects PRIVATE "${CORE}" "${ADAPTER}")
target_compile_options(r1_objects PRIVATE -O2 -Wall -Wextra -Werror -pedantic
    -fno-fast-math -ffp-contract=off -fexcess-precision=standard)
add_library(kws_r1_ref64 STATIC $<TARGET_OBJECTS:r1_objects>)
add_library(kws_r1_ref64_shared SHARED $<TARGET_OBJECTS:r1_objects>)
set_target_properties(kws_r1_ref64_shared PROPERTIES OUTPUT_NAME kws_r1_ref64 VERSION ${PROJECT_VERSION} SOVERSION 0)
foreach(t kws_r1_ref64 kws_r1_ref64_shared)
  target_link_libraries(${t} PUBLIC m)
  target_include_directories(${t} PUBLIC $<BUILD_INTERFACE:${ADAPTER}> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/kws-r1>)
endforeach()
add_executable(r1_fsmn_pcm "${ADAPTER}/r1_sdk_cli.c")
target_compile_options(r1_fsmn_pcm PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(r1_fsmn_pcm PRIVATE kws_r1_ref64)
include(CTest)
if(BUILD_TESTING)
  add_executable(r1_sdk_contracts "${ADAPTER}/test_sdk.c" "${ADAPTER}/r1_sdk.c")
  target_include_directories(r1_sdk_contracts PRIVATE "${CORE}" "${ADAPTER}")
  target_compile_options(r1_sdk_contracts PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_options(r1_sdk_contracts PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=free)
  target_link_libraries(r1_sdk_contracts PRIVATE m)
  add_test(NAME research_sdk_mock_contracts COMMAND r1_sdk_contracts)
endif()
install(TARGETS kws_r1_ref64 kws_r1_ref64_shared r1_fsmn_pcm
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${ADAPTER}/r1_sdk.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/kws-r1)
install(FILES "${CORE}/LICENSE" "${CORE}/NOTICE.md" "${CORE}/LICENSE.torchaudio"
  DESTINATION ${CMAKE_INSTALL_DATADIR}/kws-r1)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/reference64/weights/model.f32le")
  install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/reference64/weights/model.f32le"
      "${CMAKE_CURRENT_SOURCE_DIR}/reference64/weights/frontend.f32le"
      DESTINATION ${CMAKE_INSTALL_DATADIR}/kws-r1)
endif()
