function(ekws_exact_file path expected)
  if(NOT EXISTS "${path}" OR IS_SYMLINK "${path}" OR IS_DIRECTORY "${path}")
    message(FATAL_ERROR "Required regular pinned source: ${path}")
  endif()
  file(SHA256 "${path}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Frozen source byte identity mismatch: ${path}")
  endif()
endfunction()
