def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def load(p):return json.loads(p.read_text())

def require(v,m):
    if not v:raise RuntimeError(m)
def safe_local_path(root, relative):
    require(isinstance(relative,str) and relative!='', 'invalid frozen path')
    p=pathlib.PurePosixPath(relative)
    require(not p.is_absolute() and '..' not in p.parts and '.' not in p.parts,
            'escaping frozen path '+relative)
    target=root/p
    require(target.resolve().is_relative_to(root.resolve()), 'escaping resolved path '+relative)
    require(all(not (root/pathlib.Path(*p.parts[:i])).is_symlink()
                for i in range(1,len(p.parts)+1)), 'symlink frozen path '+relative)
    return target


def verify_freeze(root,freeze):
    require(freeze.get('schema')==FREEZE_SCHEMA,'wrong N0 freeze schema')
    got={}
    for x in freeze['files']:
        p=safe_local_path(root,x['path'])
        require(x['path'] not in got,'duplicate frozen path '+x['path'])
        require(p.is_file(),'missing frozen identity '+x['path'])
        require(p.stat().st_size==x['bytes'] and sha(p)==x['sha256'],
                'identity mismatch '+x['path'])
        if x.get('executable'):require(os.access(p,os.X_OK),'not executable '+x['path'])
        got[x['path']]=x['sha256']
    require(MANDATORY_FROZEN<=got.keys(),'missing mandatory N0 frozen identities')
    return got


class MissingProcField(ValueError):
    """A required field was absent, including a possible terminal status race."""


def error_record(error, path):
    """Preserve the exact exception text, rather than replacing a read with zero."""
    return {'type':type(error).__name__, 'message':str(error), 'path':str(path),
            'errno':getattr(error,'errno',None), 'filename':getattr(error,'filename',None)}


def unknown_io(reason):
    return {key:{'status':'UNKNOWN','value':None,'reason':reason} for key in IO_COUNTERS}


def read_proc(pid, read_text=None):
    """Collect critical guards separately from optional, per-counter I/O telemetry."""
    read_text = read_text or (lambda path: path.read_text())
    base=pathlib.Path('/proc')/str(pid)
    state={'critical_errors':{}, 'optional_status_errors':{}}
    status_path=base/'status'
    try:
        fields={line.split(':',1)[0]:line.split(':',1)[1].strip()
                for line in read_text(status_path).splitlines() if ':' in line}
    except (OSError,UnicodeError) as error:
        fields=None
        for key in ('VmRSS','Threads'):
            state['critical_errors'][key]=error_record(error,status_path)
    if fields is not None:
        for key in ('VmRSS','Threads','VmHWM','VmSize'):
            try:
                if key not in fields:
                    raise MissingProcField('missing '+key+' in '+str(status_path))
                parts=fields[key].split()
                if key=='Threads':
                    if len(parts)!=1 or not parts[0].isdigit() or int(parts[0])<1:
                        raise ValueError('invalid '+key+' field: '+repr(fields[key]))
                elif len(parts)!=2 or not parts[0].isdigit() or parts[1]!='kB':
                    raise ValueError('invalid '+key+' field: '+repr(fields[key]))
                state[key]=int(parts[0])
            except (ValueError,IndexError) as error:
                target='critical_errors' if key in ('VmRSS','Threads') else 'optional_status_errors'
                state[target][key]=error_record(error,status_path)

    children_path=base/'task'/str(pid)/'children'
    try:
        children=read_text(children_path).strip()
        if any(not token.isdigit() or int(token)<1 for token in children.split()):
            raise ValueError('invalid children field: '+repr(children))
        state['children']=children
    except (OSError,UnicodeError,ValueError) as error:
        state['critical_errors']['children']=error_record(error,children_path)

    io_path=base/'io'
    state['io']={}
    try:
        io_fields={line.split(':',1)[0]:line.split(':',1)[1].strip()
                   for line in read_text(io_path).splitlines() if ':' in line}
    except (OSError,UnicodeError) as error:
        for key in IO_COUNTERS:
            state['io'][key]={'status':'UNAVAILABLE','value':None,'error':error_record(error,io_path)}
    else:
        for key in IO_COUNTERS:
            try:
                if key not in io_fields:
                    raise MissingProcField('missing '+key+' in '+str(io_path))
                if not io_fields[key].isdigit():
                    raise ValueError('invalid '+key+' counter: '+repr(io_fields[key]))
                state['io'][key]={'status':'AVAILABLE','value':int(io_fields[key])}
            except ValueError as error:
                state['io'][key]={'status':'UNAVAILABLE','value':None,'error':error_record(error,io_path)}
    return state


def observe_process(proc, read_text=None):
    """Poll first. Only confirmed disappearance races relax missing terminal guards."""
    rc=proc.poll()
    if rc is not None:
        return {'returncode':rc,'observation':'EXITED_BEFORE_READ','critical_errors':{},
                'critical_monitoring':'UNKNOWN_NOT_SAMPLED_AFTER_EXIT',
                'io':unknown_io('process already exited; no post-exit read attempted')}
    state=read_proc(proc.pid,read_text)
    state['returncode']=proc.poll()
    state['observation']='LIVE' if state['returncode'] is None else 'EXITED_DURING_READ'
    errors=state['critical_errors']
    if not errors:
        state['critical_monitoring']='OBSERVED'
    elif state['returncode'] is not None and all(
            error['type'] in ('FileNotFoundError','ProcessLookupError','MissingProcField')
            for error in errors.values()):
        state['critical_monitoring']='UNKNOWN_CONFIRMED_EXIT_RACE'
    else:
        # Permission denial and malformed critical data never become success at exit.
        state['critical_monitoring']='FAILED_UNAVAILABLE'
    return state


def guard_reason(state, elapsed, output_bytes, *, limits=None, reserve=RESERVED_RECORD_BYTES):
    """Enforce known limit crossings even if this observation also sees an exit."""
    limits=LIMITS if limits is None else limits
    if elapsed>limits['wall_seconds']:return 'WALL_LIMIT'
    if 'VmRSS' in state and state['VmRSS']*1024>limits['rss_bytes']:return 'RSS_LIMIT'
    if 'Threads' in state and state['Threads']>limits['collector_threads']:return 'THREAD_LIMIT'
    if state.get('children'):return 'CHILD_PROCESS_FORBIDDEN'
    if output_bytes>limits['output_bytes']-reserve:return 'OUTPUT_LIMIT'
    if state['critical_monitoring']=='FAILED_UNAVAILABLE':return 'CRITICAL_MONITORING_UNAVAILABLE'
    return None


def apply_hard_limits(cpu, *, limits=None, file_bytes=PER_FILE_BYTES):
    limits=LIMITS if limits is None else limits
    os.sched_setaffinity(0,{cpu})
    resource.setrlimit(resource.RLIMIT_AS,(limits['rss_bytes'],)*2)
    resource.setrlimit(resource.RLIMIT_CPU,(limits['cpu_seconds'],)*2)
    resource.setrlimit(resource.RLIMIT_FSIZE,(file_bytes,)*2)
    resource.setrlimit(resource.RLIMIT_CORE,(0,0))
    signal.alarm(limits['wall_seconds'])


def monitor_process(proc, start, output_size, *, read_text=None,
                    clock=time.monotonic, sleep=time.sleep, killpg=None, record=None,
                    limits=None, reserve=RESERVED_RECORD_BYTES):
    """Injectable monitor: tests use only invented observations and a fake process."""
    killpg=os.killpg if killpg is None else killpg
    samples=[]
    peak=None
    live_samples=0
    io_last=unknown_io('no successful sample')
    record={} if record is None else record
    while True:
        state=observe_process(proc,read_text)
        elapsed=clock()-start
        if state['observation']=='LIVE' and state['critical_monitoring']=='OBSERVED':
            live_samples+=1
        for key in ('VmRSS','VmHWM'):
            if key in state:
                value=state[key]*1024
                peak=value if peak is None else max(peak,value)
        for key, counter in state['io'].items():
            if counter['status']=='AVAILABLE':
                io_last[key]={**counter,'wall_s':elapsed}
        samples.append({'wall_s':elapsed,**state})
        record.update(owned_live_guard_samples=live_samples,sampled_peak_rss_bytes=peak,io_last_successful_observation=io_last,
                      io_final_observation=state['io'],proc_samples=samples)
        reason=guard_reason(state,elapsed,output_size(),limits=limits,reserve=reserve)
        if reason is not None or state['returncode'] is not None:
            if reason is None and live_samples==0:
                reason='NO_LIVE_CRITICAL_SAMPLE'
            record['guard_stop_reason']=reason
            cleanup=cleanup_owned_process(proc,kill=(reason is not None),killpg=killpg)
            record.update(cleanup=cleanup,returncode=cleanup['returncode'])
            if cleanup['errors'] and reason is None:
                record['guard_stop_reason']='PROCESS_CLEANUP_FAILED'
            return record
        sleep(0.1)


def encode_resource_record(record, artifact_bytes):
    """Bound the final write too; oversized diagnostics fail rather than overrun."""
    def encode(value):
        return (json.dumps(value,sort_keys=True,indent=2,allow_nan=False)+'\n').encode()
    def fits(data):
        return len(data)<=RESERVED_RECORD_BYTES and artifact_bytes+len(data)<=LIMITS['output_bytes']
    record=dict(record)
    data=encode(record)
    if fits(data):return record,data
    samples=record['proc_samples']
    record.update(status='FAILED_NO_RETRY',
                  guard_stop_reason=record.get('guard_stop_reason') or 'RESOURCE_RECORD_OUTPUT_LIMIT',
                  resource_record_output_limit=True,proc_samples_total=len(samples),
                  proc_samples_omitted=max(0,len(samples)-2),
                  proc_samples=samples if len(samples)<=2 else [samples[0],samples[-1]])
    data=encode(record)
    # Never write beyond the cap, including when even reduced failure evidence cannot fit.
    return record,data if fits(data) else None


def cleanup_owned_process(proc, *, kill=True, killpg=None):
    """Never replace a primary failure with a cleanup exception or wait forever.

    The group belongs to the child created with start_new_session=True. Kill its
    group even if the leader just exited: an observed forbidden descendant may
    still be present. ESRCH is preserved as an exit race, never as success proof.
    """
    killpg=os.killpg if killpg is None else killpg
    result={'returncode':None,'kill_requested':kill,'kill_exit_race':False,
            'reaped':False,'errors':[]}
    if proc is None:return result
    if kill:
        try:killpg(proc.pid,signal.SIGKILL)
        except ProcessLookupError as error:
            result.update(kill_exit_race=True,kill_exit_race_evidence=error_record(error,'owned process group'))
        except BaseException as error:
            result['errors'].append(error_record(error,'kill owned process group'))
    try:
        result.update(returncode=proc.wait(timeout=1),reaped=True)
    except BaseException as error:
        result['errors'].append(error_record(error,'wait owned process'))
        result['returncode']=getattr(proc,'returncode',None)
    return result


def exception_evidence(error,proc,record, *, killpg=None):
    """Both normal guards and this outer exceptional path retain failure evidence."""
    record['supervisor_exception']=error_record(error,'supervisor')
    record['guard_stop_reason']=record.get('guard_stop_reason') or 'SUPERVISOR_EXCEPTION:'+type(error).__name__+':'+str(error)
    cleanup=cleanup_owned_process(proc,kill=True,killpg=killpg)
    record.update(cleanup=cleanup,returncode=cleanup['returncode'])
    return record


def validate_release_header(release, *, probe=False):
    schema='a20-n0-guard-probe-release-v1' if probe else RELEASE_SCHEMA
    field='owner_guard_probe_release' if probe else 'owner_execution_release'
    words='EXECUTE_EXACT_N0_HARMLESS_GUARD_PROBE_ONCE' if probe else EXECUTION_WORDS
    require(release.get('schema')==schema,'N0-specific release schema required; old Qwen release forbidden')
    require(release.get(field)==words,'explicit N0 parent release required')
    require(release.get('no_competing_acoustic_run_confirmed') is True,'resource context not confirmed')


def check_receipt(ref, schema, binding, *, scope=None):
    path=pathlib.Path(ref['path'])
    require(path.is_absolute() and path.is_file() and not path.is_symlink(),'receipt must be an absolute regular file')
    require(sha(path)==ref['sha256'],'receipt identity mismatch '+str(path))
    receipt=load(path)
    require(receipt.get('schema')==schema,'receipt schema mismatch '+str(path))
    require(receipt.get('status')=='PASS','receipt is not PASS '+str(path))
    for key,value in binding.items():require(receipt.get(key)==value,'receipt binding mismatch '+key)
    if scope is not None:require(receipt.get('scope')==scope,'receipt scope mismatch')
    return receipt


def binding_for(root):
    return {'freeze_sha256':sha(root/'SOURCE-FREEZE.json'),
            'protocol_sha256':sha(root/'PROTOCOL.json'),
            'launcher_sha256':sha(root/'src/launch_once.py'),
            'guard_probe_sha256':sha(root/'src/guard_probe.py')}


def verify_system_dependencies(protocol):
    dependencies=protocol['system_dependencies']
    python=pathlib.Path(sys.executable).resolve()
    require(str(python) in dependencies,'executing Python not frozen')
    for path,value in dependencies.items():
        p=pathlib.Path(path)
        require(p.is_absolute() and p.is_file(),'missing system dependency '+path)
        require(sha(p)==value,'system dependency identity mismatch '+path)
    return python


def validate_protocol(protocol,identities,binding):
    require(protocol.get('schema')==PROTOCOL_SCHEMA,'wrong N0 protocol schema')
    require(protocol.get('limits')==LIMITS,'limits drift')
    require(protocol.get('output_budget')==OUTPUT_BUDGET,'output budget drift')
    require(sum(OUTPUT_BUDGET.values())==LIMITS['output_bytes'],'invalid aggregate output allocation')
    for key,value in {'input_count':3,'passes':1,'warmups':0,'retries':0}.items():
        require(type(protocol.get(key)) is int and protocol[key]==value,'execution scope drift '+key)
    for key in ('launcher_sha256','guard_probe_sha256'):
        require(protocol.get(key)==binding[key],'protocol code binding mismatch '+key)
    for name,key in [('build/collector','collector_sha256'),
                     ('metadata/verified-inputs.json','manifest_sha256'),
                     ('metadata/geometry.json','geometry_sha256'),
                     ('metadata/decoder-config.json','decoder_config_sha256')]:
        require(protocol.get(key)==identities[name],'protocol frozen identity mismatch '+key)
    require(protocol.get('guard_probe')=={'scope':PROBE_SCOPE,'limits':PROBE_LIMITS,
        'file_bytes':PROBE_FILE_BYTES,'command':PROBE_COMMAND,
        'python_flags':['-I','-S'],'required_live_samples':1},'guard probe contract drift')
    require(type(protocol.get('cpu_affinity')) is int,'invalid CPU affinity')


def admission(root,release_path, *, probe=False):
    # Header gate precedes any lock, ledger, output creation or child launch.
    release=load(release_path)
    validate_release_header(release,probe=probe)
    require(root.name=='a20-n0-deterministic-negative-controls-v1','wrong N0 root')
    binding=binding_for(root)
    for key in ('freeze_sha256','protocol_sha256'):
        require(release.get(key)==binding[key],'release identity mismatch '+key)
    protocol=load(root/'PROTOCOL.json');freeze=load(root/'SOURCE-FREEZE.json')
    identities=verify_freeze(root,freeze)
    require(identities['src/launch_once.py']==sha(pathlib.Path(__file__)),
            'executing supervisor is not exact frozen source')
    validate_protocol(protocol,identities,binding)
    python=verify_system_dependencies(protocol)
    cpu=protocol['cpu_affinity']
    require(cpu in os.sched_getaffinity(0),'frozen CPU unavailable')
    common={key:binding[key] for key in ('freeze_sha256','protocol_sha256')}
    if probe:
        check_receipt(release['independent_guard_probe_review'],
                      'a20-n0-independent-guard-probe-review-v1',binding,scope=PROBE_SCOPE)
        check_receipt(release['private_snapshot_readback'],'a20-n0-private-snapshot-readback-v1',common)
    else:
        check_receipt(release['independent_review'],'a20-n0-independent-preparation-review-v1',common)
        check_receipt(release['private_snapshot_readback'],'a20-n0-private-snapshot-readback-v1',common)
        report=check_receipt(release['guard_probe_report'],'a20-n0-live-guard-probe-report-v1',binding,scope=PROBE_SCOPE)
        require(type(report.get('candidate_runs')) is int and report['candidate_runs']==0
                and type(report.get('probe_processes_attempted')) is int and report['probe_processes_attempted']==1,
                'invalid probe execution counts')
        require(type(report.get('owned_live_guard_samples')) is int and report['owned_live_guard_samples']>=1,
                'probe never observed live critical fields')
        require(report.get('guard_stop_reason') is None and report.get('returncode')==0,
                'failed guard probe cannot admit collector')
        require(report.get('identity_after_ok') is True,'probe identity drift')
        require(report.get('limits')==PROBE_LIMITS and report.get('file_bytes')==PROBE_FILE_BYTES,
                'probe limit drift')
        require(report.get('python_sha256')==sha(python),'probe Python identity drift')
        require(report.get('host_uname')==list(os.uname()),'probe host mismatch')
        require(report.get('boot_id_sha256')==sha(pathlib.Path('/proc/sys/kernel/random/boot_id')),
                'probe host boot mismatch')
        # Re-check the exact static probe review bound into the runtime report.
        check_receipt(report['independent_guard_probe_review'],
                      'a20-n0-independent-guard-probe-review-v1',binding,scope=PROBE_SCOPE)
    return release,protocol,freeze,identities,binding,python


def encoded(value):
    return (json.dumps(value,sort_keys=True,indent=2,allow_nan=False)+'\n').encode()


def output_size(out):
    total=0
    for p in out.iterdir():
        require(p.is_file() and not p.is_symlink(),'unexpected run output entry '+str(p))
        total+=p.stat().st_size
    return total


def write_exclusive(path,data, *, cap):
    require(len(data)<=cap,'encoded output exceeds file allocation '+str(path))
    with open(path,'xb') as stream:
        stream.write(data);stream.flush();os.fsync(stream.fileno())


def write_metadata(out,ledger,name,value, *, replace_ledger=False):
    data=encoded(value)
    existing=sum(p.stat().st_size for p in out.iterdir()
                 if p.name not in ('raw.jsonl','stderr.log','resource-record.json'))
    existing+=0 if replace_ledger else ledger.stat().st_size
    require(existing+len(data)<=OUTPUT_BUDGET['metadata_bytes'],'metadata output allocation exceeded')
    if replace_ledger:
        with open(ledger,'wb') as stream:
            stream.write(data);stream.flush();os.fsync(stream.fileno())
    else:
        write_exclusive(out/name,data,cap=OUTPUT_BUDGET['metadata_bytes'])


def reserve_attempt(root,release_path,binding, *, probe=False):
    # O_EXCL consumption is permanent even when later setup or launch fails.
    ledger=root/('GUARD-PROBE-LEDGER.json' if probe else 'EXECUTION-LEDGER.json')
    out=root/('guard-probe-once' if probe else 'run-once')
    record={'schema':'a20-n0-attempt-ledger-v1','status':'START_RESERVED_NO_RETRY',
            'kind':'GUARD_PROBE' if probe else 'CANDIDATE','release_sha256':sha(release_path),
            **binding,'utc_epoch_s':time.time(),'once_only':True,'retries_allowed':False}
    fd=os.open(ledger,os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o600)
    with os.fdopen(fd,'wb') as stream:
        stream.write(encoded(record));stream.flush();os.fsync(stream.fileno())
    out.mkdir(mode=0o700,exist_ok=False)
    return ledger,out


def clean_environment():
    # No LD/PYTHON startup injection and no inherited environment outside this allowlist.
    return {'PATH':'/usr/bin:/bin','LANG':'C','LC_ALL':'C','OMP_NUM_THREADS':'1',
            'OPENBLAS_NUM_THREADS':'1','MKL_NUM_THREADS':'1','NUMEXPR_NUM_THREADS':'1'}


def identity_after(root,freeze,protocol,identities,binding):
    try:
        require(binding_for(root)==binding,'post-run freeze or supervisor identity drift')
        after=verify_freeze(root,freeze)
        verify_system_dependencies(protocol)
        require(after==identities,'post-run identity drift')
        return after,True
    except Exception as error:return {'error':error_record(error,'post-run identities')},False


def observer_usage():
    """Unbounded separate supervisor overhead, never charged as child CPU."""
    usage=resource.getrusage(resource.RUSAGE_SELF)
    return {'process_cpu_ns':time.process_time_ns(),'user_cpu_s':usage.ru_utime,
            'system_cpu_s':usage.ru_stime,'maxrss_bytes_linux':usage.ru_maxrss*1024}


def observer_evidence(before):
    after=observer_usage()
    return {'before':before,'after':after,
            'process_cpu_ns_delta':after['process_cpu_ns']-before['process_cpu_ns'],
            'user_cpu_s_delta':after['user_cpu_s']-before['user_cpu_s'],
            'system_cpu_s_delta':after['system_cpu_s']-before['system_cpu_s'],
            'scope':'RUN_ENTRY_BEFORE_ADMISSION_THROUGH_BEFORE_FINAL_RECORD_ENCODING',
            'includes':'admission and final identity hashing, child launch/wait, proc monitoring, earlier metadata writes',
            'excludes':'Python startup/imports and final resource record/ledger encoding and persistence',
            'maxrss_is_process_lifetime_high_water_not_delta':True,
            'child_hard_limits_do_not_bound_observer':True}


def run(release_path):
    observer_before=observer_usage()
    release,protocol,freeze,identities,binding,python=admission(ROOT,release_path)
    shared=ROOT.parent/'PUBLIC_SHARED_LOCK_LOCATION_NOT_INCLUDED'
    with open(shared,'a+') as lock:
        fcntl.flock(lock,fcntl.LOCK_EX|fcntl.LOCK_NB)
        # Revalidate after lock and before permanently consuming the sole attempt.
        require(verify_freeze(ROOT,freeze)==identities,'pre-reservation identity drift')
        ledger,out=reserve_attempt(ROOT,release_path,binding)
        start=time.monotonic();proc=None
        observation={'sampled_peak_rss_bytes':None,'owned_live_guard_samples':0,
                     'io_last_successful_observation':unknown_io('no successful sample'),
                     'io_final_observation':unknown_io('no observation'),'proc_samples':[],
                     'returncode':None,'guard_stop_reason':None}
        try:
            write_metadata(out,ledger,'release-used.json',release)
            write_metadata(out,ledger,'identities-before.json',identities)
            write_metadata(out,ledger,'resource-context.json',{
                'host_uname':list(os.uname()),'cpu_affinity':[protocol['cpu_affinity']],
                'supervisor_separate_from_single_collector_process':True,
                'parallel_context':release.get('parallel_context'),'limits':LIMITS,
                'output_budget':OUTPUT_BUDGET,'clock_monotonic_resolution_s':time.get_clock_info('monotonic').resolution})
            argv=[str(ROOT/'build/collector'),'--run',binding['protocol_sha256'],
                  identities['metadata/verified-inputs.json'],identities['metadata/geometry.json'],
                  identities['metadata/decoder-config.json']]
            def guard():apply_hard_limits(protocol['cpu_affinity'])
            with open(out/'raw.jsonl','xb',buffering=0) as raw,open(out/'stderr.log','xb',buffering=0) as err:
                proc=subprocess.Popen(argv,cwd=ROOT,stdin=subprocess.DEVNULL,stdout=raw,stderr=err,
                                      env=clean_environment(),start_new_session=True,preexec_fn=guard)
                monitor_process(proc,start,lambda:output_size(out)+ledger.stat().st_size,record=observation)
        except BaseException as error:
            exception_evidence(error,proc,observation)
        after,identity_ok=identity_after(ROOT,freeze,protocol,identities,binding)
        try:write_metadata(out,ledger,'identities-after.json',after)
        except BaseException as error:
            observation['evidence_write_error']=error_record(error,'identities-after.json')
            observation['guard_stop_reason']=observation.get('guard_stop_reason') or 'IDENTITY_EVIDENCE_WRITE_FAILED'
        status='RAW_COMPLETE_PENDING_AUDIT' if observation['returncode']==0 and observation['guard_stop_reason'] is None and identity_ok else 'FAILED_NO_RETRY'
        artifact_bytes=output_size(out)+ledger.stat().st_size
        raw_sha256=sha(out/'raw.jsonl') if (out/'raw.jsonl').exists() else None
        own_usage=observer_evidence(observer_before)
        record,data=encode_resource_record({'schema':'a20-n0-resource-record-v1','status':status,
            **binding,**observation,'elapsed_wall_s':time.monotonic()-start,
            'sampling_interval_s':0.1,'sampling_may_miss_exit_peak':True,
            'exact_child_rusage_in_raw_run_end_if_complete':True,'identity_after_ok':identity_ok,
            'raw_sha256':raw_sha256,'observer_usage':own_usage,
            'artifact_bytes_before_resource_record_write':artifact_bytes,
            'candidate_processes_attempted':int(proc is not None),'candidate_launch_attempted':True,
            'probe_processes_attempted':0,'retries_allowed':False},artifact_bytes)
        try:
            if data is not None:write_exclusive(out/'resource-record.json',data,cap=RESERVED_RECORD_BYTES)
            else:record['status']='FAILED_NO_RETRY'
        except BaseException as error:
            record['status']='FAILED_NO_RETRY'
            record['guard_stop_reason']=record.get('guard_stop_reason') or 'RESOURCE_EVIDENCE_WRITE_FAILED'
            record['resource_write_error']=error_record(error,'resource-record.json')
        write_metadata(out,ledger,None,{'schema':'a20-n0-attempt-ledger-v1','status':record['status'],
            'release_sha256':sha(release_path),**binding,'output_dir':'run-once',
            'returncode':record['returncode'],'guard_stop_reason':record['guard_stop_reason'],
            'cleanup':record.get('cleanup'),'supervisor_exception':record.get('supervisor_exception'),
            'resource_write_error':record.get('resource_write_error'),'once_only':True,'retries_allowed':False},replace_ledger=True)
        return 0 if record['status']=='RAW_COMPLETE_PENDING_AUDIT' else 1
